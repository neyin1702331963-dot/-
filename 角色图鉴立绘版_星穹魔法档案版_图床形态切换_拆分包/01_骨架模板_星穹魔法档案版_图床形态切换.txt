(function(){

        var a=s.parentNode,F=[];
        var c=s.closest?s.closest('[data-forms]'):(a&&a.parentNode);
        if(!a||!c){return false;}
        try{var _r=c.getAttribute('data-forms')||'[]';_r=_r.replace(/&quot;/g,'"').replace(/&#34;/g,'"').replace(/&amp;/g,'&');F=JSON.parse(_r)}catch(err){}
        if(!F.length){return false;}
        var i=((parseInt(c.getAttribute('data-form-index')||'0',10)||0)+1)%F.length;
        var f=F[i]||{};
        c.setAttribute('data-form-index',i);
        c.setAttribute('data-current-form',f.name||'');
        a.setAttribute('data-form-caption',f.role||f.name||'');
        var m=a.querySelector('img');
        if(m&&f.src){
            m.src=f.src;
            m.alt=(c.getAttribute('data-char-name')||'')+' - '+(f.name||'');
        }
        var r=c.querySelector('.custom-char-role');
        if(r){r.textContent=f.role||'';}
        var q=c.querySelector('.custom-char-quote');
        if(q){q.textContent=f.quote||'';}
        var e=c.querySelector('.custom-char-event');
        if(e){
            e.textContent=f.event||'';
            var b=document.createElement('b');
            b.textContent='形态记录';
            e.insertBefore(b,e.firstChild);
        }
        var t=c.querySelector('.custom-char-tags');
        if(t){
            t.textContent='';
            (f.tags||[]).forEach(function(x){
                var n=document.createElement('span');
                n.textContent=x;
                t.appendChild(n);
            });
        }
        s.textContent=(i+1)+'/'+F.length;
        a.classList.remove('custom-art-switching');
        a.classList.add('custom-art-switched');
        setTimeout(function(){a.classList.remove('custom-art-switched');},520);
        return false;
    };
    document.addEventListener('click',function(ev){
        var s=ev.target&&ev.target.closest?ev.target.closest('.custom-form-switch[data-cg-switch]'):null;
        if(!s){return;}
        if(ev.preventDefault){ev.preventDefault();}
        if(ev.stopPropagation){ev.stopPropagation();}
        window.cgSwitchFormBtn(s);
    });
