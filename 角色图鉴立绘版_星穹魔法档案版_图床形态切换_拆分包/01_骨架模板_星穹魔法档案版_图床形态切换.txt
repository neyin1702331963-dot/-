(function(){
    function parseForms(c){

    function apply(art,card,f,i,F){
        card.setAttribute('data-form-index',i);
        card.setAttribute('data-current-form',f.name||'');
        art.setAttribute('data-form-caption',f.role||f.name||'');
        var m=art.querySelector('img');
        if(m&&f.src){m.src=f.src;m.alt=(card.getAttribute('data-char-name')||'')+' - '+(f.name||'');}

        var q=card.querySelector('.custom-char-quote');if(q){q.textContent=f.quote||'';}
        var e=card.querySelector('.custom-char-event');

        if(t){t.textContent='';(f.tags||[]).forEach(function(x){var sp=document.createElement('span');sp.textContent=x;t.appendChild(sp);});}
        var sw=card.querySelector('.custom-form-switch');if(sw){sw.textContent=(i+1)+'/'+F.length;}
    }
    function switchForm(art){
        if(!art||art.__cgBusy){return false;}
        var card=art.closest?art.closest('[data-forms]'):null;
        if(!card){return false;}
        var F=parseForms(card);
        if(!F.length){return false;}
        art.__cgBusy=true;
        var i=((parseInt(card.getAttribute('data-form-index')||'0',10)||0)+1)%F.length,f=F[i]||{};

        art.classList.add('custom-art-switching');

            apply(art,card,f,i,F);

        },210);

    }
